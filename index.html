// EVENT HANDLER // ========================================== function onFormSubmit(e) { try { const itemResponses = e.response.getItemResponses(); let studentFirstName = ''; let studentLastName = ''; let incidentDetails = []; // Parse Form Responses for (let i = 0; i < itemResponses.length; i++) { const response = itemResponses[i]; const title = response.getItem().getTitle().toLowerCase().trim(); const value = response.getResponse(); if (title.includes('first name') || title.includes('student first')) { studentFirstName = value.toString().trim(); } else if (title.includes('last name') || title.includes('student last')) { studentLastName = value.toString().trim(); } else { const formattedVal = Array.isArray(value) ? value.join(', ') : value; incidentDetails.push(`${response.getItem().getTitle()}: ${formattedVal}`); } } // Match Student to Case Manager const matchedStudent = findStudent(studentFirstName, studentLastName); if (matchedStudent) { sendEmailNotification(matchedStudent.cm, `${matchedStudent.fn} ${matchedStudent.ln}`, incidentDetails); } else { Logger.log(`No matching Case Manager found for: ${studentFirstName} ${studentLastName}`); sendEmailNotification( ADMIN_FALLBACK_EMAIL, `${studentFirstName} ${studentLastName} (Unmatched)`, incidentDetails ); } } catch (err) { Logger.log(`Error processing form submission: ${err.message}`); } } // ========================================== // HELPER FUNCTIONS // ========================================== function findStudent(firstName, lastName) { if (!firstName || !lastName) return null; const cleanFN = firstName.toLowerCase().trim(); const cleanLN = lastName.toLowerCase().trim().replace('.', ''); return MASTER_ROSTER.find(student => { const rosterFN = student.fn.toLowerCase().trim(); const rosterLN = student.ln.toLowerCase().trim().replace('.', ''); return rosterFN === cleanFN && rosterLN.startsWith(cleanLN.charAt(0)); }); } ${incidentDetails.join('

This is an automated notification from the Behavior Reporting System.

`; GmailApp.sendEmail(recipientEmail, subject, '', { htmlBody: htmlBody }); }
